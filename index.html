/**
 * BeautyVisionService
 * Extracts structured product data from photos using a vision model.
 *
 * Prompt engineering techniques:
 * — Constraints BEFORE permissions
 * — Uncertainty anchoring
 * — 5-block prompt structure
 */
final readonly class BeautyVisionService
{
    private const string VISION_PROMPT = <<<PROMPT
    [ROLE]
    You are a beauty product data extractor for an e-commerce platform.
    Think step by step in English. Output ONLY valid JSON.

    [CONSTRAINTS]
    FORBIDDEN — violating any rule makes the response invalid:
    - Invent data not clearly visible in the image
    - Guess skin type if not printed on label
    - Infer country of origin from brand name — only read "Made in ..."
    - Fill ingredient list from memory — read from photo only

    [BEHAVIOR]
    - Read ALL text visible in photos carefully
    - If a field value is not clearly visible → return null for that field
    - Numeric values must include units (ml, %, SPF number)

    [FIELDS]
    brandName, productLine, volumeMl, spfFactor,
    keyIngredients (array), skinType, countryOfOrigin,
    ean, conditionGrade ("New" or "Good")

    [FORMAT]
    Return ONLY JSON object. No markdown. No explanation.
    PROMPT;

    public function __construct(
        private HttpClientInterface $httpClient,
        private LoggerInterface    $logger,
        private string              $visionModel,
    ) {}

    /** @param string[] $imagePaths Local paths to product photos */
    public function extractFromPhotos(array $imagePaths): array
    {
        $images = array_map(
            fn($path) => [
                'type'      => 'image_url',
                'image_url' => ['url' => 'data:image/jpeg;base64,'
                    . base64_encode(file_get_contents($path))],
            ],
            $imagePaths
        );

        $messages = [[
            'role'    => 'user',
            'content' => [
                ...$images,
                ['type' => 'text', 'text' => self::VISION_PROMPT],
            ],
        ]];

        try {
            $response = $this->httpClient->request('POST', '/api/chat', [
                'json'    => [
                    'model'    => $this->visionModel,
                    'messages' => $messages,
                    'options'  => [
                        'temperature' => 0.1,  // low — factual extraction
                        'num_predict' => 800,
                    ],
                ],
                'timeout' => 90,
            ]);

            $raw = $response->toArray()['message']['content'] ?? '';
            return $this->parseJson($raw);

        } catch (\Throwable $e) {
            $this->logger->error('Vision extraction failed: {err}', [
                'err' => $e->getMessage(),
            ]);
            return [];
        }
    }

    private function parseJson(string $raw): array
    {
        $clean = preg_replace('/```(?:json)?\s*|\s*```/', '', $raw);
        if (preg_match('/\{.*\}/s', trim($clean), $m)) {
            $data = json_decode($m[0], true);
            return is_array($data) ? $data : [];
        }
        return [];
    }
}
/**
 * BeautyEnrichmentService
 * Generates GEO-optimized description based on E-GEO research patterns.
 *
 * Best performing strategies applied:
 * — Competitive  +1.61 avg rank improvement
 * — FAQ intent   +1.20
 * — Factuality preserved in 12/15 optimized prompts
 *
 * Generates ONLY: description, texture, conditionGrade
 * Never generates: certifications, compatibleSkinTypes, countryOfOrigin
 */
final readonly class BeautyEnrichmentService
{
    public function __construct(
        private AiModelClientInterface $aiClient,
        private LoggerInterface        $logger,
    ) {}

    public function enrich(array $data): array
    {
        $empty = $this->detectEmptyFields($data);

        if (empty($empty) || !$this->aiClient->isAvailable()) {
            return [];
        }

        try {
            $prompt = $this->buildGeoPrompt($data, $empty);
            $raw    = $this->aiClient->generateText($prompt, temperature: 0.5);
            return $this->filterEmpty($this->parseJson($raw), $empty);
        } catch (\RuntimeException $e) {
            $this->logger->error('Enrichment failed: {err}', ['err' => $e->getMessage()]);
            return [];
        }
    }

    private function detectEmptyFields(array $data): array
    {
        // Only fields safe from hallucination
        return array_values(array_filter(
            ['description', 'texture', 'conditionGrade'],
            fn($f) => empty(trim($data[$f] ?? ''))
        ));
    }

    private function buildGeoPrompt(array $data, array $fields): string
    {
        // Build context from all available extracted fields
        $ingredients = implode(', ', $data['keyIngredients'] ?? []);
        $ctx = implode("\n", array_filter([
            !empty($data['brandName'])     ? "Brand: {$data['brandName']}"           : null,
            !empty($data['productLine'])   ? "Product: {$data['productLine']}"         : null,
            !empty($data['volumeMl'])      ? "Volume: {$data['volumeMl']} ml"          : null,
            !empty($data['spfFactor'])     ? "SPF: {$data['spfFactor']}"               : null,
            !empty($data['skinType'])      ? "Skin type: {$data['skinType']}"          : null,
            !empty($ingredients)           ? "Key ingredients: {$ingredients}"         : null,
            !empty($data['countryOfOrigin'])? "Origin: {$data['countryOfOrigin']}"      : null,
            !empty($data['ean'])       ? "EAN: {$data['ean']}"                 : null,
        ]));

        // GEO field definitions — E-GEO Competitive + FAQ pattern
        $defs = [
            'description'   => '"description": "GEO text for AI search engines (ChatGPT, Perplexity, Google SGE).
              Structure: WHAT+FOR WHOM → active ingredients → texture → SPF → skin benefit → EAN.
              Include ALL numeric parameters. Address buyer intent: what skin problem does it solve?
              Competitive differentiation: why this product over alternatives?
              5–10 sentences. No adjectives: reliable/perfect/best/luxurious are forbidden."',
            'texture'        => '"texture": "cream / gel-cream / serum / lightweight emulsion / rich balm"',
            'conditionGrade' => '"conditionGrade": "New or Good only"',
        ];

        $jsonFields = implode(",\n    ",
            array_map(fn($f) => $defs[$f] ?? '', $fields)
        );

        return <<<PROMPT
        You are a GEO copywriter for a beauty e-commerce store.
        Think in English. Respond in English only.

        FORBIDDEN:
        - Invent ingredients or clinical claims not present in the data
        - Marketing words: "luxurious", "perfect", "best", "amazing", "holy grail"
        - Generic phrases: "suitable for all skin types" unless stated in data
        - Mention price, discounts, or stock availability

        Product data:
        {$ctx}

        Fill ONLY these fields. Respond with strict JSON, no markdown:
        {
            {$jsonFields}
        }
        PROMPT;
    }

    private function filterEmpty(array $ai, array $empty): array
    {
        return array_filter(
            array_intersect_key($ai, array_flip($empty)),
            fn($v) => !empty($v)
        );
    }

    private function parseJson(string $raw): array
    {
        $clean = preg_replace('/```(?:json)?\s*|\s*```/', '', $raw);
        if (preg_match('/\{.*\}/s', trim($clean), $m)) {
            $data = json_decode($m[0], true);
            return is_array($data) ? $data : [];
        }
        return [];
    }
}
/**
 * BeautyChatService
 * Streaming chat for interactive product card refinement.
 *
 * Prompt engineering techniques:
 * — Critical first tokens — output starts with [field:name]
 * — Disambiguation — one clarifying question if ambiguous
 * — 5-block structure
 */
final readonly class BeautyChatService
{
    private const string SYSTEM_PROMPT = <<<PROMPT
    [ROLE]
    Technical copywriter for a beauty e-commerce platform.
    Help the seller fill in the product card for a face cream.
    Think in English internally. Output in English only.

    [CONSTRAINTS]
    FORBIDDEN:
    - Invent ingredients, clinical studies, or certifications not provided
    - Use words: "luxurious", "perfect", "best", "amazing", "holy grail"
    - Suggest price, discount, or stock quantity
    - Fill certifications, compatibleSkinTypes — those come from catalog

    [BEHAVIOR]
    - On greeting: briefly describe capabilities. Do NOT generate data.
    - Generate fields ONLY when seller explicitly requests it
    - If seller request is ambiguous → ask ONE clarifying question (CPR)
    - Keep responses concise and factual

    [FIELD_RULES]
    description:
    - GEO structure: WHAT+FOR WHOM → ingredients → texture → SPF → benefit → EAN
    - All numeric parameters included (ml, %, SPF number)
    - 5–10 sentences. Concrete. No marketing language.
    texture:
    - cream / gel-cream / serum / lightweight emulsion / rich balm
    conditionGrade:
    - "New" or "Good" only — no other values

    [FORMAT]
    Field output: [field:name]value[/field]
    PROMPT;

    public function __construct(
        private HttpClientInterface $httpClient,
        private LoggerInterface    $logger,
        private string              $textModel,
    ) {}

    public function chat(BeautyChatRequest $request): StreamedResponse
    {
        $system   = $this->buildSystemWithContext($request->cardContext);
        $messages = $this->buildMessages($system, $request->messages);

        return new StreamedResponse(
            fn() => $this->stream($messages),
            200,
            [
                'Content-Type'      => 'text/event-stream',
                'Cache-Control'     => 'no-cache',
                'X-Accel-Buffering' => 'no',
            ]
        );
    }

    private function buildSystemWithContext(?array $ctx): string
    {
        if (empty($ctx)) {
            return self::SYSTEM_PROMPT;
        }

        $map = [
            'brandName'       => 'Brand',
            'productLine'     => 'Product',
            'volumeMl'        => 'Volume (ml)',
            'spfFactor'       => 'SPF',
            'keyIngredients'  => 'Key ingredients',
            'skinType'        => 'Skin type',
            'texture'         => 'Texture',
            'countryOfOrigin' => 'Country of origin',
            'ean'         => 'EAN / Barcode',
        ];

        $lines = [];
        foreach ($map as $key => $label) {
            $val = $ctx[$key] ?? null;
            if (!empty($val)) {
                $lines[] = "{$label}: "
                    . (is_array($val) ? implode(', ', $val) : $val);
            }
        }

        return self::SYSTEM_PROMPT
            . "\n\n[CARD_DATA]\n"
            . implode("\n", $lines);
    }

    private function stream(array $messages): void
    {
        try {
            $response = $this->httpClient->request('POST', '/api/chat', [
                'json' => [
                    'model'    => $this->textModel,
                    'messages' => $messages,
                    'stream'   => true,
                    'options'  => [
                        'temperature'    => 0.5,
                        'top_p'          => 0.95,
                        'repeat_penalty' => 1.1,
                        'num_predict'    => 1500,
                    ],
                ],
                'timeout' => 120,
                'buffer'  => false,
            ]);

            foreach ($this->httpClient->stream($response) as $chunk) {
                foreach (explode("\n", trim($chunk->getContent())) as $line) {
                    $d = json_decode($line, true);
                    if (!is_array($d)) continue;

                    $token = $d['message']['content'] ?? '';
                    if ($token !== '') {
                        echo "data: " . json_encode(['token' => $token]) . "\n\n";
                        ob_flush(); flush();
                    }
                    if ($d['done'] ?? false) {
                        echo "data: [DONE]\n\n";
                        ob_flush(); flush();
                        return;
                    }
                }
            }
        } catch (\Throwable $e) {
            $this->logger->error('Chat stream error: {err}', ['err' => $e->getMessage()]);
            echo "data: " . json_encode(['error' => 'AI service unavailable']) . "\n\n";
            ob_flush(); flush();
        }
    }

    private function buildMessages(string $sys, array $msgs): array
    {
        $out = [['role' => 'system', 'content' => $sys]];
        foreach ($msgs as $m) {
            if (!empty($m['content'])) {
                $out[] = [
                    'role'    => $m['role'] === 'assistant' ? 'assistant' : 'user',
                    'content' => $m['content'],
                ];
            }
        }
        return $out;
    }
}
